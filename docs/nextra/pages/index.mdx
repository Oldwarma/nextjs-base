## 🚀 快速开始

只需 **5 分钟**，创建你的第一个 CRUD 管理页面：

### 1. 创建 Server Actions

```javascript filename="app/(admin)/actions/rbac/crud-action.product.js"
'use server';

import { wrapQueryAction } from '@/lib/core/action-wrapper';

export const getProductListAction = wrapQueryAction('product', async (params) => {
  const { pageIndex = 1, pageSize = 20, whereJson = {}, sortJson = {} } = params;
  
  const { getCollection } = await import('@/lib/database/mongodb');
  const collection = await getCollection('products');
  
  const skip = (pageIndex - 1) * pageSize;
  const [data, total] = await Promise.all([
    collection.find(whereJson).sort(sortJson).skip(skip).limit(pageSize).toArray(),
    collection.countDocuments(whereJson),
  ]);
  
  return { success: true, data, total, page: pageIndex, pageSize };
});

export async function createProductAction(data) {
  // 创建逻辑
}

export async function updateProductAction(id, data) {
  // 更新逻辑
}

export async function deleteProductAction(id) {
  // 删除逻辑
}
```

### 2. 创建前端页面

```javascript filename="app/(admin)/admin/products/page.js"
'use client';

import { useMemo } from 'react';
import dynamic from 'next/dynamic';
import * as productActions from '@/app/(admin)/actions/rbac/crud-action.product';

const SmartCrudPage = dynamic(() => import('@/components/admin/smart-crud-page'), {
  ssr: false,
});

export default function ProductsPage() {
  const fieldsConfig = useMemo(() => [
    {
      key: 'name',
      title: 'Product Name',
      type: 'text',
      table: { width: 200, copyable: true },
      form: { required: true },
      search: { enabled: true, mode: 'like' },
    },
    {
      key: 'price',
      title: 'Price',
      type: 'number',
      table: { width: 120, sorter: true },
      form: { required: true },
    },
    {
      key: 'status',
      title: 'Status',
      type: 'select',
      options: [
        { label: 'Active', value: 'active', color: 'green' },
        { label: 'Inactive', value: 'inactive', color: 'red' },
      ],
      search: { enabled: true },
    },
  ], []);

  return (
    <SmartCrudPage
      fieldsConfig={fieldsConfig}
      actions={productActions}
      title="Product Management"
      rowKey="id"
      enableCreate
      enableEdit
      enableDelete
    />
  );
}
```

### 3. 完成！

访问 `/admin/products`，你的 CRUD 管理页面已经完成！
