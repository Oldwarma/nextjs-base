## 🎯 概述

NextJS Base 使用 **next-intl** 实现国际化，支持多语言路由和翻译。

### 支持的语言

| 语言 | 代码 | 文件 |
|:---|:---|:---|
| 英语 | `en` | `i18n/messages/en.json` |
| 中文 | `zh` | `i18n/messages/zh.json` |
| 日语 | `ja` | `i18n/messages/ja.json` |

---

## ⚙️ 配置说明

### 目录结构

```
i18n/
├── config.js           # 配置文件
├── request.js          # 请求配置
└── messages/           # 翻译文件
    ├── en.json
    ├── zh.json
    └── ja.json
```

### 配置文件

```javascript
// i18n/config.js
export const locales = ['en', 'zh', 'ja']
export const defaultLocale = 'en'

export const localeNames = {
  en: 'English',
  zh: '中文',
  ja: '日本語',
}
```

### 请求配置

```javascript
// i18n/request.js
import { getRequestConfig } from 'next-intl/server'

export default getRequestConfig(async ({ locale }) => ({
  messages: (await import(`./messages/${locale}.json`)).default,
}))
```

### Next.js 配置

```javascript
// next.config.mjs
import createNextIntlPlugin from 'next-intl/plugin'

const withNextIntl = createNextIntlPlugin('./i18n/request.js')

export default withNextIntl({
  // 其他配置
})
```

## 🔧 使用方法

### 客户端组件

```javascript
'use client'

import { useTranslations } from 'next-intl'

export default function HomePage() {
  const t = useTranslations('home')
  const tCommon = useTranslations('common')

  return (
    <div>
      <h1>{t('title')}</h1>
      <p>{t('description')}</p>
      <button>{tCommon('confirm')}</button>
    </div>
  )
}
```

### 服务端组件

```javascript
import { getTranslations } from 'next-intl/server'

export default async function HomePage() {
  const t = await getTranslations('home')

  return (
    <div>
      <h1>{t('title')}</h1>
      <p>{t('description')}</p>
    </div>
  )
}
```

### Server Actions

```javascript
'use server'

import { getTranslations } from 'next-intl/server'

export async function submitFormAction(data) {
  const t = await getTranslations('common')
  
  try {
    // 处理逻辑
    return { success: true, message: t('success') }
  } catch (error) {
    return { success: false, message: t('error') }
  }
}
```

### 带参数的翻译

```json
// messages/en.json
{
  "greeting": "Hello, {name}!",
  "items": "You have {count, plural, =0 {no items} one {# item} other {# items}}"
}
```

```javascript
const t = useTranslations()

// 简单参数
t('greeting', { name: 'John' })  // "Hello, John!"

// 复数处理
t('items', { count: 0 })   // "You have no items"
t('items', { count: 1 })   // "You have 1 item"
t('items', { count: 5 })   // "You have 5 items"
```

### 富文本翻译

```json
{
  "terms": "By signing up, you agree to our <link>Terms of Service</link>"
}
```

```javascript
t.rich('terms', {
  link: (chunks) => <a href="/terms">{chunks}</a>
})
```

## 🌐 语言切换

### 语言切换组件

```javascript
// components/common/LanguageSwitcher.jsx
'use client'

import { useLocale } from 'next-intl'
import { useRouter, usePathname } from 'next/navigation'
import { locales, localeNames } from '@/i18n/config'

export default function LanguageSwitcher() {
  const locale = useLocale()
  const router = useRouter()
  const pathname = usePathname()

  const handleChange = (newLocale) => {
    // 替换路径中的语言代码
    const newPath = pathname.replace(`/${locale}`, `/${newLocale}`)
    router.push(newPath)
  }

  return (
    <select value={locale} onChange={(e) => handleChange(e.target.value)}>
      {locales.map((loc) => (
        <option key={loc} value={loc}>
          {localeNames[loc]}
        </option>
      ))}
    </select>
  )
}
```

### 简单版本

```javascript
// components/common/LanguageSwitcherSimple.jsx
'use client'

import { useLocale } from 'next-intl'
import Link from 'next/link'
import { usePathname } from 'next/navigation'
import { locales, localeNames } from '@/i18n/config'

export default function LanguageSwitcherSimple() {
  const locale = useLocale()
  const pathname = usePathname()

  return (
    <div className="flex gap-2">
      {locales.map((loc) => {
        const newPath = pathname.replace(`/${locale}`, `/${loc}`)
        const isActive = locale === loc

        return (
          <Link
            key={loc}
            href={newPath}
            className={isActive ? 'font-bold' : ''}
          >
            {localeNames[loc]}
          </Link>
        )
      })}
    </div>
  )
}
```

## 📁 路由结构

### 前台路由

```
app/(client)/
└── [locale]/
    ├── layout.js           # 带语言的布局
    ├── page.js             # 首页
    ├── auth/
    │   ├── login/page.js
    │   └── register/page.js
    └── [其他页面]/page.js
```

### 布局配置

```javascript
// app/(client)/[locale]/layout.js
import { NextIntlClientProvider } from 'next-intl'
import { getMessages } from 'next-intl/server'

export default async function LocaleLayout({ children, params: { locale } }) {
  const messages = await getMessages()

  return (
    <NextIntlClientProvider locale={locale} messages={messages}>
      {children}
    </NextIntlClientProvider>
  )
}
```

## ✅ 最佳实践

### 1. 翻译文件组织

```json
{
  "common": {
    // 通用文本
  },
  "auth": {
    // 认证相关
  },
  "home": {
    // 首页
  },
  "product": {
    // 产品页面
  },
  "error": {
    // 错误消息
  }
}
```

### 2. 命名规范

```
✅ 推荐：语义化命名
{
  "auth": {
    "loginButton": "Login",
    "loginSuccess": "Login successful",
    "loginError": "Login failed"
  }
}

❌ 不推荐：无意义命名
{
  "auth": {
    "text1": "Login",
    "text2": "Login successful"
  }
}
```

### 3. 避免硬编码

```javascript
// ✅ 推荐：使用翻译
const t = useTranslations('common')
<button>{t('submit')}</button>

// ❌ 不推荐：硬编码
<button>Submit</button>
```

### 4. 保持一致性

确保所有语言文件的结构一致：

```bash
# 检查翻译完整性
npm run i18n:check  # 如果有此脚本
```
